GUIA DE CONFIGURAÇÃO SQL SUPABASE - APPCAL
========================================================================

Copia o script abaixo e cola-o no teu editor SQL do Supabase.
(Menu lateral esquerdo > SQL Editor > New Query > Run)

------------------------------------------------------------------------
1. SCRIPT DE CRIAÇÃO DAS TABELAS E SEGURANÇA
------------------------------------------------------------------------

-- Habilitar a extensão UUID
create extension if not exists "uuid-ossp";

-- TABELA 1: weight_logs (Registo de Peso Corporal)
create table if not exists public.weight_logs (
    id uuid default gen_random_uuid() primary key,
    user_id uuid references auth.users(id) on delete cascade not null,
    date date not null default current_date,
    weight_kg numeric(5,2) not null,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null,
    unique(user_id, date)
);

-- TABELA 2: meals (Refeições)
create table if not exists public.meals (
    id uuid default gen_random_uuid() primary key,
    user_id uuid references auth.users(id) on delete cascade not null,
    meal_type text not null,
    logged_at timestamp with time zone not null default now(),
    photos text[] default '{}'::text[],
    total_calories integer not null,
    total_protein numeric(5,1) not null,
    total_carbs numeric(5,1) not null,
    total_fats numeric(5,1) not null,
    notes text,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- TABELA 3: meal_items (Ingredientes da Refeição)
create table if not exists public.meal_items (
    id uuid default gen_random_uuid() primary key,
    meal_id uuid references public.meals(id) on delete cascade not null,
    name text not null,
    weight_g numeric(6,1) not null,
    calories integer not null,
    protein numeric(5,1) not null,
    carbs numeric(5,1) not null,
    fats numeric(5,1) not null
);

-- TABELA 4: favorites (Modelos Guardados / Atalhos 1-Clique)
create table if not exists public.favorites (
    id uuid default gen_random_uuid() primary key,
    user_id uuid references auth.users(id) on delete cascade not null,
    name text not null,
    items jsonb not null,
    total_calories integer not null,
    total_protein numeric(5,1) not null,
    total_carbs numeric(5,1) not null,
    total_fats numeric(5,1) not null,
    created_at timestamp with time zone default timezone('utc'::text, now()) not null
);

-- Habilitar Row Level Security (RLS)
alter table public.weight_logs enable row level security;
alter table public.meals enable row level security;
alter table public.meal_items enable row level security;
alter table public.favorites enable row level security;

-- Criar políticas de acesso individual
create policy "Utilizadores acedem aos seus próprios logs de peso"
    on public.weight_logs for all using (auth.uid() = user_id);

create policy "Utilizadores acedem às suas próprias refeições"
    on public.meals for all using (auth.uid() = user_id);

create policy "Utilizadores acedem aos itens das suas refeições"
    on public.meal_items for all using (
        exists (
            select 1 from public.meals 
            where public.meals.id = public.meal_items.meal_id 
            and public.meals.user_id = auth.uid()
        )
    );

create policy "Utilizadores acedem aos seus próprios favoritos"
    on public.favorites for all using (auth.uid() = user_id);


------------------------------------------------------------------------
2. QUERIES DE VERIFICAÇÃO (Executar no SQL Editor para testar)
------------------------------------------------------------------------

A. Confirmar se as 4 tabelas estão no ar:
-----------------------------------------
select table_name 
from information_schema.tables 
where table_schema = 'public' 
and table_name in ('meals', 'meal_items', 'weight_logs', 'favorites');

(Deve retornar meals, meal_items, weight_logs e favorites).


B. Confirmar se o RLS (Segurança) está ativo (deve retornar true em tudo):
------------------------------------------------------------------------
select tablename, rowsecurity 
from pg_tables 
where schemaname = 'public' 
and tablename in ('meals', 'meal_items', 'weight_logs', 'favorites');


C. Consultar contagem geral de dados na app:
--------------------------------------------
select 
  (select count(*) from public.meals) as total_refeicoes,
  (select count(*) from public.meal_items) as total_ingredientes,
  (select count(*) from public.weight_logs) as total_pesagens,
  (select count(*) from public.favorites) as total_favoritos;
